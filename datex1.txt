import pandas as pd

data = {
    "Date": [
        "2026-08-01", "2026-08-02", "2026-08-03", "2026-08-04", "2026-08-05",
        "2026-08-06", "2026-08-07", "2026-08-08", "2026-08-09", "2026-08-10"
    ],
    "Product": [
        "Laptop", "Smartphone", "Headphones", "Keyboard", "Mouse",
        "Monitor", "Laptop", "Smartphone", "Headphones", "Keyboard"
    ],
    "Sales": [75000, 48000, 12000, 7500, 5000, 36000, 50000, 72000, 15000, 9000],
    "Quantity": [3, 4, 8, 10, 20, 6, 2, 6, 10, 12],
    "Place": [
        "Chennai", "Bangalore", "Coimbatore", "Chennai", "Madurai",
        "Bangalore", "Hyderabad", "Chennai", "Salem", "Coimbatore"
    ]
}

# Group by product and calculate total values and quantity

import pandas as pd

data = {
    "Date": [
        "2026-08-01", "2026-08-02", "2026-08-03", "2026-08-04", "2026-08-05",
        "2026-08-06", "2026-08-07", "2026-08-08", "2026-08-09", "2026-08-10"
    ],
    "Product": [
        "Laptop", "Smartphone", "Headphones", "Keyboard", "Mouse",
        "Monitor", "Laptop", "Smartphone", "Headphones", "Keyboard"
    ],
    "Sales": [
        75000, 48000, 12000, 7500, 5000,
        36000, 50000, 72000, 15000, 9000
    ],
    "Quantity": [3, 4, 8, 10, 20, 6, 2, 6, 10, 12],
    "Place": [
        "Chennai", "Bangalore", "Coimbatore", "Chennai", "Madurai",
        "Bangalore", "Hyderabad", "Chennai", "Salem", "Coimbatore"
    ]
}

df = pd.DataFrame(data)


product_summary = df.groupby("Product")[["Sales", "Quantity"]].sum()


print("\nTOTAL SALES AND QUANTITY BY PRODUCT")
print("-----------------------------------")
print(product_summary)
